{{- $labels := dict "committee-ordinary" (slice) "committee-substitute" (slice) "company-ordinary" (slice) "company-substitute" (slice) "other-ordinary" (slice) "other-substitute" (slice) -}} {{- range .assignments -}} {{- $assignment := . -}} {{- $groups := where site.Params.roster.groups "id" $assignment.group -}} {{- if not $groups -}} {{- errorf "Unknown roster group %q for member %q" $assignment.group $.name -}} {{- else -}} {{- $group := index $groups 0 -}} {{- $label := $group.id -}} {{- if $assignment.board -}} {{- if (index site.Params.roster.boards $assignment.board) -}} {{- $label = printf "%s %s" (index site.Params.roster.boards $assignment.board).abbr $label -}} {{- else -}} {{- errorf "Unknown board %q for member %q" $assignment.board $.name -}} {{- end -}} {{- end -}} {{- if not (in (slice "ordinary" "substitute") $assignment.seat) -}} {{- errorf "Unknown seat %q for roster group %q and member %q" $assignment.seat $assignment.group $.name -}} {{- else -}} {{- $category := "" -}} {{- if eq $group.type "committee" -}} {{- $category = "committee" -}} {{- else if eq $group.type "company" -}} {{- $category = "company" -}} {{- else if in (slice "other" "reference") $group.type -}} {{- $category = "other" -}} {{- else -}} {{- errorf "Unknown roster group type %q for group %q" $group.type $assignment.group -}} {{- end -}} {{- with $assignment.title }}{{ $label = printf "%s (%s)" $label . }}{{ end -}} {{- $key := printf "%s-%s" $category $assignment.seat -}} {{- $current := index $labels $key -}} {{- $current = $current | append $label -}} {{- $labels = merge $labels (dict $key $current) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $labels -}}
