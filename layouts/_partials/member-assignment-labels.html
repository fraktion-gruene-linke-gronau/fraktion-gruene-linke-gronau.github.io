{{- $labels := dict "ordinary" (slice) "substitute" (slice) -}} {{- range $index, $assignment := .assignments -}} {{- $groups := where site.Params.roster.groups "id" $assignment.group -}} {{- if not $groups -}} {{- errorf "Unknown roster group %q for member %q" $assignment.group $.name -}} {{- else -}} {{- $group := index $groups 0 -}} {{- $label := $group.id -}} {{- if $assignment.board -}} {{- if (index site.Params.roster.boards $assignment.board) -}} {{- $label = printf "%s %s" (index site.Params.roster.boards $assignment.board).abbr $label -}} {{- else -}} {{- errorf "Unknown board %q for member %q" $assignment.board $.name -}} {{- end -}} {{- end -}} {{- if not (in (slice "ordinary" "substitute") $assignment.seat) -}} {{- errorf "Unknown seat %q for roster group %q and member %q" $assignment.seat $assignment.group $.name -}} {{- else -}} {{- with $assignment.title }}{{ $label = printf "%s (%s)" $label . }}{{ end -}} {{- $hint := or $group.name $group.id -}} {{- if $assignment.board -}} {{- $board := index site.Params.roster.boards $assignment.board -}} {{- $hint = printf "%s: %s" $board.name $group.name -}} {{- end -}} {{- $peers := partial "member-assignment-peers.html" (dict "group" $assignment.group "board" ($assignment.board | default "") "member" $.name ) -}} {{- $item := dict "label" $label "hint" $hint "peers" $peers -}} {{- $current := index $labels $assignment.seat -}} {{- $current = $current | append $item -}} {{- $labels = merge $labels (dict $assignment.seat $current) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $labels -}}
